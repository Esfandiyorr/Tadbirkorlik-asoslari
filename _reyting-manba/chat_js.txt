/* ================== TerDU CHAT ================== */
var CH=(function(){
"use strict";
var CFG=window.RT_CFG||{};
var open=false, msgs=[], keys=[], last=0, seen=0, timer=null, busy=false;
function T(uz,en){ var l=(document.documentElement.getAttribute("lang")||"uz"); return (l==="en"&&en)?en:uz; }
function esc(x){ return String(x==null?"":x).replace(/[<>&"]/g,function(c){return {"<":"&lt;",">":"&gt;","&":"&amp;",'"':"&quot;"}[c];}); }
function el(h){ var d=document.createElement("div"); d.innerHTML=h.trim(); return d.firstChild; }
function base(){ var c=window.RT&&RT.cloud?RT.cloud():null; return (c&&c.on)?c.base:null; }
function me(){ return window.RT&&RT.student?RT.student():null; }
function isTc(){ return !!(window.RT&&RT.teacher&&RT.teacher()); }

/* ---------- qoidalar nazorati ---------- */
var BAD=("jalab,kot,qoti,qotoq,skay,onangni,onasini,otangni,padar,haromi,harom zoda,ahmoq,jinni,tentak,eshak,"+
 "bler,blya,blyat,suka,pizd,huy,xuy,ebat,ebal,nahuy,pidr,mudak,gandon,dolbo,zaeb,"+
 "fuck,fck,shit,bitch,asshole,bastard,dick,cunt,whore,slut,nigger,retard").split(",");
function norm(s){
  return s.toLowerCase()
    .replace(/[ʻʼ'`’‘]/g,"")
    .replace(/[4@]/g,"a").replace(/[0]/g,"o").replace(/[1!|]/g,"i").replace(/[3]/g,"e").replace(/[5$]/g,"s")
    .replace(/(.)\1{2,}/g,"$1$1")
    .replace(/[^a-zЀ-ӿ ]+/g," ");
}
function badWord(s){
  var n=norm(s), flat=n.replace(/ +/g,"");
  for(var i=0;i<BAD.length;i++){
    var b=BAD[i].trim(); if(b.length<3) continue;
    if(n.indexOf(b)>=0||flat.indexOf(b.replace(/ /g,""))>=0) return true;
  }
  return false;
}
function hasImage(s){
  return /data:image/i.test(s) || /https?:\/\/\S+\.(jpg|jpeg|png|gif|webp|bmp|svg|mp4|webm)/i.test(s) ||
         /<\s*img/i.test(s);
}

/* ---------- bulut ---------- */
function pull(cb){
  var b=base(); if(!b){ cb&&cb(false); return; }
  fetch(b+'/chat.json?orderBy="$key"&limitToLast=80').then(function(r){ return r.ok?r.json():null; })
  .then(function(v){
    var a=[],k=[];
    if(v){ Object.keys(v).sort().forEach(function(key){ if(v[key]&&v[key].t){ a.push(v[key]); k.push(key); } }); }
    msgs=a; keys=k; cb&&cb(true);
  }).catch(function(){ cb&&cb(false); });
}
function push(rec,cb){
  var b=base(); if(!b){ cb(false); return; }
  fetch(b+"/chat.json",{method:"POST",headers:{"Content-Type":"application/json"},body:JSON.stringify(rec)})
    .then(function(r){ cb(r.ok); }).catch(function(){ cb(false); });
}
function del(key){
  var b=base(); if(!b) return;
  fetch(b+"/chat/"+key+".json",{method:"DELETE"}).then(function(){ pull(draw); }).catch(function(){});
}

/* ---------- interfeys ---------- */
function hhmm(t){ var d=new Date(t||0); function z(n){return (n<10?"0":"")+n;} return z(d.getHours())+":"+z(d.getMinutes()); }
function draw(){
  var list=document.getElementById("chList"); if(!list) return;
  var S=me();
  if(!msgs.length){
    list.innerHTML='<p class="ch-empty">'+T("Hozircha xabar yo\'q. Birinchi bo\'lib yozing 👋","No messages yet. Be the first to write 👋")+'</p>';
  }else{
    list.innerHTML=msgs.map(function(m,i){
      var mine=S&&m.id===S.id;
      return '<div class="ch-msg'+(mine?" me":"")+'"><div class="w"><span class="n">'+esc(m.n)+'</span>'+
        '<span class="g">'+esc(m.g||"")+'</span><span class="tm">'+hhmm(m.at)+'</span></div>'+esc(m.t)+
        (isTc()?'<button class="del" data-k="'+esc(keys[i])+'" title="'+T("O\'chirish","Delete")+'">🗑</button>':'')+'</div>';
    }).join("");
    list.querySelectorAll(".del").forEach(function(b){ b.onclick=function(){ del(b.dataset.k); }; });
    list.scrollTop=list.scrollHeight;
  }
  badge();
}
function badge(){
  var d=document.getElementById("chDot"); if(!d) return;
  if(open){ seen=msgs.length; try{localStorage.setItem("terdu_chat_seen",seen);}catch(e){} d.classList.remove("on"); return; }
  var n=msgs.length-seen;
  if(n>0){ d.textContent=n>99?"99+":n; d.classList.add("on"); } else d.classList.remove("on");
}
function warn(txt){
  var w=document.getElementById("chWarn"); if(!w) return;
  if(!txt){ w.classList.remove("on"); return; }
  w.textContent=txt; w.classList.add("on");
  clearTimeout(w._h); w._h=setTimeout(function(){ w.classList.remove("on"); },5000);
}
function send(){
  var ta=document.getElementById("chTa"); if(!ta) return;
  var v=ta.value.replace(/\s+/g," ").trim();
  if(!v) return;
  if(!(window.RT&&RT.ok&&RT.ok())){ warn(T("Avval o'qituvchi sizni tasdiqlashi kerak.","The teacher must approve you first.")); return; }
  if(v.length>300){ warn(T("Xabar juda uzun (300 belgigacha).","The message is too long (300 characters max).")); return; }
  if(hasImage(v)){ warn(T("⛔ Rasm va video havolalari taqiqlangan — faqat matn yozing.","⛔ Image and video links are not allowed — text only.")); return; }
  if(badWord(v)){ warn(T("⛔ Haqoratli so'z aniqlandi. Iltimos, hurmat bilan yozing.","⛔ Offensive language detected. Please write respectfully.")); return; }
  if(Date.now()-last<2500){ warn(T("Biroz sekinroq yozing.","Please write a bit more slowly.")); return; }
  var S=me(); if(!S) return;
  busy=true; document.getElementById("chSend").disabled=true;
  push({id:S.id,n:S.name+" "+S.surname,g:S.group,t:v,at:Date.now()},function(ok){
    busy=false; document.getElementById("chSend").disabled=false;
    if(!ok){ warn(T("Yuborilmadi — internetni tekshiring.","Not sent — check your connection.")); return; }
    ta.value=""; last=Date.now(); warn("");
    pull(function(){ seen=msgs.length; draw(); });
  });
}
function build(){
  if(document.getElementById("chTab")) return;
  var t=el('<button class="ch-tab" id="chTab" title="'+T("Talabalar chati","Student chat")+'">💬<i>CHAT</i><span class="ch-dot" id="chDot"></span></button>');
  var p=el('<aside class="ch-panel" id="chPanel">'+
    '<div class="ch-hd"><b>💬 '+T("Talabalar chati","Student chat")+'</b><button class="ch-x" id="chX">✕</button></div>'+
    '<div class="ch-rules"><b>📌 '+T("Chat qoidalari","Chat rules")+'</b>'+
      '<u>⛔ '+T("So\'kinish, haqorat va qo\'pol gaplar taqiqlanadi.","No swearing, insults or rude language.")+'</u>'+
      '<u>⛔ '+T("Yomon rasm, video va havolalar yuborish mumkin emas.","Do not post bad images, videos or links.")+'</u>'+
      '<u>✅ '+T("Faqat dars va tadbirkorlik mavzusida muloqot qiling.","Keep the conversation about the lesson and entrepreneurship.")+'</u>'+
      '<u>👨‍🏫 '+T("O\'qituvchi chatni ko\'rib boradi: qoidabuzar bloklanadi.","The teacher monitors the chat: rule-breakers are blocked.")+'</u></div>'+
    '<div class="ch-list" id="chList"></div>'+
    '<p class="ch-warn" id="chWarn"></p>'+
    '<div class="ch-in"><textarea id="chTa" maxlength="300" placeholder="'+T("Xabar yozing…","Write a message…")+'"></textarea>'+
      '<button class="ch-send" id="chSend" title="'+T("Yuborish","Send")+'">➤</button></div>'+
  '</aside>');
  document.body.appendChild(t); document.body.appendChild(p);
  try{ seen=parseInt(localStorage.getItem("terdu_chat_seen")||"0",10)||0; }catch(e){}
  t.onclick=toggle;
  document.getElementById("chX").onclick=toggle;
  document.getElementById("chSend").onclick=send;
  document.getElementById("chTa").addEventListener("keydown",function(e){
    if(e.key==="Enter"&&!e.shiftKey){ e.preventDefault(); send(); }
  });
  if(!base()){
    document.getElementById("chList").innerHTML='<p class="ch-lock">🔌 '+T(
      "Chat serverga ulanmagan. O'qituvchi bazani ulagach, chat ishlaydi.",
      "The chat is not connected to the server. It will work once the teacher connects the database.")+'</p>';
    document.getElementById("chTa").disabled=true; document.getElementById("chSend").disabled=true;
    return;
  }
  pull(function(){ if(!seen) seen=msgs.length; draw(); });
  tick();
  document.addEventListener("visibilitychange",function(){ if(!document.hidden) refresh(); });
}
function refresh(){ pull(function(ok){ if(ok) draw(); }); }
function tick(){
  clearTimeout(timer);
  timer=setTimeout(function(){ if(!document.hidden&&!busy) refresh(); tick(); }, open?5000:25000);
}
function toggle(){
  var p=document.getElementById("chPanel"); if(!p) return;
  open=!open; p.classList.toggle("show",open);
  if(open){ refresh(); seen=msgs.length; badge(); setTimeout(function(){ var l=document.getElementById("chList"); if(l) l.scrollTop=l.scrollHeight; },60); }
  tick();
}
function langSync(){
  var tab=document.getElementById("chTab"); if(!tab) return;
  var was=open; tab.remove(); var p=document.getElementById("chPanel"); if(p) p.remove();
  build(); if(was){ open=false; toggle(); }
}
function wait(){
  if(window.RT_ready&&window.RT&&RT.ok&&RT.ok()){
    build();
    var lb=document.getElementById("btnLang");
    if(lb) lb.addEventListener("click",function(){ setTimeout(langSync,40); });
    return;
  }
  setTimeout(wait,900);
}
if(document.readyState==="loading") document.addEventListener("DOMContentLoaded",wait); else wait();
return {open:function(){ if(!open) toggle(); }, refresh:refresh};
})();
