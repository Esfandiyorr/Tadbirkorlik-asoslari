function bases(){

function probe(b,cb){
  var done=false, t=setTimeout(function(){ if(!done){done=true;cb(0);} },5000);
  fetch(b+"/students.json?shallow=true",{method:"GET"})
    .then(function(r){ if(!done){done=true;clearTimeout(t);cb(r.status);} })
    .catch(function(){ if(!done){done=true;clearTimeout(t);cb(0);} });
}
function cloudInit(done){
  var f=CFG.firebase;
  if(!f||(!f.db&&!f.url)||typeof fetch!=="function"){ done(false); return; }
  var saved=null; try{ saved=localStorage.getItem("terdu_base"); }catch(e){}
  var list=bases();
  if(saved && list.indexOf(saved)<0) list.unshift(saved);
  else if(saved){ list.splice(list.indexOf(saved),1); list.unshift(saved); }
  var i=0;
  (function next(){
    if(i>=list.length){ cloudErr=cloudErr||"topilmadi"; done(false); return; }
    var b=list[i++];
    probe(b,function(st){
      if(st===200){ BASE=b; cloudOn=true; try{localStorage.setItem("terdu_base",b);}catch(e){} done(true); }
      else if(st===401||st===403){ BASE=b; cloudErr="qoida"; done(false); }
      else next();
    });
  })();
}

function okMail(x){ return /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(String(x||"").trim()); }
function kb(n){ return n>=1048576 ? (n/1048576).toFixed(1)+" MB" : Math.max(1,Math.round(n/1024))+" KB"; }

/* -- o'qituvchi Firebase Auth orqali kiradi; shundagina CV o'qiy oladi -- */
var TA=lget("terdu_ta",null);
function taSave(){ lset("terdu_ta",TA); }
function taOut(){ TA=null; lset("terdu_ta",null); }
function taOn(){ return !!(TA&&TA.key&&TA.rt); }
function taFresh(cb){
  if(!taOn()){ cb(false); return; }
  if(TA.tok&&Date.now()<(TA.exp||0)-60000){ cb(true); return; }
  fetch("https://securetoken.googleapis.com/v1/token?key="+encodeURIComponent(TA.key),{
    method:"POST",headers:{"Content-Type":"application/x-www-form-urlencoded"},
    body:"grant_type=refresh_token&refresh_token="+encodeURIComponent(TA.rt)})
   .then(function(r){ return r.ok?r.json():null; })
   .then(function(j){
     if(!j||!j.id_token){ cb(false); return; }
     TA.tok=j.id_token; TA.rt=j.refresh_token||TA.rt;
     TA.exp=Date.now()+(parseInt(j.expires_in||3600,10)*1000); taSave(); cb(true);
   }).catch(function(){ cb(false); });
}
function taLogin(key,em,pw,cb){
  fetch("https://identitytoolkit.googleapis.com/v1/accounts:signInWithPassword?key="+encodeURIComponent(key),{
    method:"POST",headers:{"Content-Type":"application/json"},
    body:JSON.stringify({email:em,password:pw,returnSecureToken:true})})
   .then(function(r){ return r.json().then(function(j){ return {ok:r.ok,j:j}; }); })
   .then(function(x){
     if(!x.ok||!x.j.idToken){ cb(false,(x.j&&x.j.error&&x.j.error.message)||"XATO"); return; }
     TA={key:key,email:em,uid:x.j.localId,tok:x.j.idToken,rt:x.j.refreshToken,
         exp:Date.now()+(parseInt(x.j.expiresIn||3600,10)*1000)};
     taSave(); cb(true,x.j.localId);
   }).catch(function(){ cb(false,"tarmoq"); });
}

function cvSend(f,onProg,cb){
  if(!cloudOn||!BASE||!S){ cb(false,T("Baza ulanmagan.","No database connection.")); return; }
  var ext=(f.name.split(".").pop()||"").toLowerCase();
  if(CVOK.indexOf(ext)<0){ cb(false,T("Faqat PDF, DOC, DOCX yoki rasm bo'lishi mumkin.","Only PDF, DOC, DOCX or an image is allowed.")); return; }
  if(f.size>CVMAX){ cb(false,T("Fayl juda katta: ","File too large: ")+kb(f.size)+" ("+T("chegara","limit")+" "+kb(CVMAX)+")"); return; }
  var fr=new FileReader();
  fr.onerror=function(){ cb(false,T("Faylni o'qib bo'lmadi.","Could not read the file.")); };
  fr.onload=function(){
    onProg&&onProg(35);
    var meta={n:f.name,t:f.type||ext,s:f.size,at:Date.now(),
              who:S.name+" "+S.surname,gr:S.group,em:S.email||""};
    var full={}; for(var k in meta) full[k]=meta[k]; full.d=fr.result;
    fetch(BASE+"/cv/"+S.id+".json",{method:"PUT",headers:{"Content-Type":"application/json"},body:JSON.stringify(full)})
     .then(function(r){
       if(!r.ok){ cb(false,(r.status===401||r.status===403)
         ?T("Qoidalar ruxsat bermadi. O'qituvchiga ayting.","The rules denied the upload. Tell your teacher."):("HTTP "+r.status)); return; }
       onProg&&onProg(85);
       /* ommaviy belgi: fayl emas, faqat "yuklangan" ma'lumoti */
       return fetch(BASE+"/students/"+S.id+"/cv.json",{method:"PUT",headers:{"Content-Type":"application/json"},
         body:JSON.stringify({n:meta.n,s:meta.s,at:meta.at})}).then(function(){
           onProg&&onProg(100);
           S.cv={n:meta.n,s:meta.s,at:meta.at}; lset(LS.stu,S);
           cb(true,"");
         });
     }).catch(function(){ cb(false,T("Tarmoq xatosi.","Network error.")); });
  };
  fr.readAsDataURL(f);
}

function topicSummary(){
  var o={};
  for(var t in P.topics){ o[t]={p:P.topics[t].p||0,s:Object.keys(P.topics[t].slides||{}).length}; }
  return o;
}
function topicObj(){
  var t=CFG.topic||"m0";
  if(!P.topics[t]) P.topics[t]={p:0,slides:{},acts:{}};
  return P.topics[t];
}
function award(key,pts,label){
  if(!S||!approved()) return;
  var t=topicObj(), full=(CFG.topic||"m0")+":"+key;
  if(P.log[full]) return;
  P.log[full]=pts; t.p=(t.p||0)+pts; P.total=(P.total||0)+pts;
  if(key.indexOf("slide")===0) t.slides[key]=1; else t.acts[key]=pts;
  save(); chipUpdate(); toast("+"+pts+" ball"+(label?" · "+label:""));
}
function slideSeen(n){
  award("slide"+n,CFG.pSlide||2,"slayd");
  var t=topicObj(), seen=Object.keys(t.slides).length;
  if(CFG.slides && seen>=CFG.slides) award("allSlides",CFG.pAll||10,"mavzu to'liq");
}

function dupGroups(){
  var by={}, out=[];
  tcList.forEach(function(x){
    var k=((x.name||"")+" "+(x.surname||"")).toLowerCase().replace(/\s+/g," ").trim();
    (by[k]=by[k]||[]).push(x);
  });
  for(var k in by) if(by[k].length>1) out.push(by[k]);
  return out;
}
function tcDedupe(){
  var gr=dupGroups(); if(!gr.length) return;
  var kill=[], names=[];
  gr.forEach(function(a){
    a.sort(function(x,y){                       /* eng ko'p ballik, keyin eng yangisi qoladi */
      return (y.total||0)-(x.total||0) || (y.updated||y.at||0)-(x.updated||x.at||0);
    });
    names.push(a[0].name+" "+a[0].surname+" ("+(a.length-1)+")");
    kill=kill.concat(a.slice(1));
  });
  if(!confirm(T("Takroriy yozuvlar o'chirilsinmi","Delete duplicate records")+"?\n"+names.join("\n")+
     "\n\n"+T("Har biridan eng ko'p ballisi qoldiriladi.","The one with the most points is kept for each."))) return;
  kill.forEach(function(x){ cloudDrop(x.id); });
  var ids=kill.map(function(x){return x.id;});
  tcList=tcList.filter(function(x){ return ids.indexOf(x.id)<0; });
  tcDraw(); toast(T("Tozalandi","Cleaned")+": "+kill.length);
  if(S&&ids.indexOf(S.id)>=0){ lset(LS.stu,null); setTimeout(function(){ location.reload(); },700); }
}

function tcCvGet(id,nm,btn){
  if(!cloudOn||!BASE){ toast(T("Baza ulanmagan","Not connected")); return; }
  var t0=btn.innerHTML; btn.disabled=true; btn.innerHTML="⏳";
  cvUrl("/cv/"+id,function(u){
    fetch(u).then(function(r){
      if(!r.ok){
        btn.disabled=false; btn.innerHTML=t0;
        toast((r.status===401||r.status===403)
          ? T("Ruxsat yo'q — pastdagi \"Maxfiy kirish\" bo'limidan kiring.","No access — sign in under \"Private access\" below.")
          : ("HTTP "+r.status));
        return null;
      }
      return r.json();
    }).then(function(v){
      btn.disabled=false; btn.innerHTML=t0;
      if(!v) return;
      if(!v.d){ toast(T("Fayl topilmadi","File not found")); return; }
      var a=document.createElement("a"); a.href=v.d; a.download=(v.n||nm||"cv");
      document.body.appendChild(a); a.click(); a.remove();
      toast(T("CV yuklab olindi","CV downloaded"));
    }).catch(function(){ btn.disabled=false; btn.innerHTML=t0; toast(T("Tarmoq xatosi","Network error")); });
  });
}
function mails(list){
  var seen={}, out=[];
  list.forEach(function(x){
    var e=String(x.email||"").trim().toLowerCase();
    if(okMail(e)&&!seen[e]){ seen[e]=1; out.push(e); }
  });
  return out;
}

function tcMailBind(okl){
  var gm=document.getElementById("mlGm"); if(!gm) return;
  var m=mails(okl);
  function txt(){
    return {s:(document.getElementById("mlSub")||{}).value||"",
            b:(document.getElementById("mlBody")||{}).value||""};
  }
  function note(t){ var n=document.getElementById("mlMsg"); if(n) n.textContent=t; }
  function guard(){
    if(!m.length){ note(T("Hech kimda email yo'q.","Nobody has an email yet.")); return false; }
    return true;
  }
  /* uzun ro'yxatni bo'laklarga bo'lamiz — havola uzunligi cheklangan */
  function parts(){
    var out=[], step=45;
    for(var i=0;i<m.length;i+=step) out.push(m.slice(i,i+step));
    return out;
  }
  function open1(mk){
    if(!guard()) return;
    var t=txt(), ps=parts();
    ps.forEach(function(chunk,i){
      setTimeout(function(){ window.open(mk(chunk,t),"_blank"); },i*700);
    });
    note(ps.length>1
      ? T("Ro'yxat uzun — "+ps.length+" ta oyna ochiladi (har birida 45 tagacha manzil).",
          ps.length+" windows will open (up to 45 addresses each).")
      : T("Pochta oynasi ochildi. Yuborishni o'zingiz tasdiqlaysiz.","The mail window is open. You send it yourself."));
  }
  gm.onclick=function(){ open1(function(c,t){
    return "https://mail.google.com/mail/?view=cm&fs=1&to=&bcc="+encodeURIComponent(c.join(","))+
      "&su="+encodeURIComponent(t.s)+"&body="+encodeURIComponent(t.b); }); };
  document.getElementById("mlMt").onclick=function(){ open1(function(c,t){
    return "mailto:?bcc="+encodeURIComponent(c.join(","))+
      "&subject="+encodeURIComponent(t.s)+"&body="+encodeURIComponent(t.b); }); };
  document.getElementById("mlCp").onclick=function(){
    if(!guard()) return;
    var str=m.join(", ");
    function done(){ note(T("Nusxalandi: ","Copied: ")+m.length+" "+T("ta manzil","addresses")); toast(T("Nusxalandi ✅","Copied ✅")); }
    if(navigator.clipboard&&navigator.clipboard.writeText){
      navigator.clipboard.writeText(str).then(done,function(){ note(str); });
    } else { note(str); }
  };
}
